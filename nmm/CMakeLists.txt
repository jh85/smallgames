cmake_minimum_required(VERSION 3.16)
project(morris CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_FLAGS_RELEASE "-O3 -march=native -funroll-loops")
add_compile_options(-Wall -Wextra)
add_library(morris_core src/board.cpp)
target_include_directories(morris_core PUBLIC src)
add_library(morris_zdd src/zdd1.cpp src/zdd2.cpp)
target_link_libraries(morris_zdd morris_core)
add_executable(test_board tests/test_board.cpp)
target_link_libraries(test_board morris_core)
enable_testing()
add_executable(test_zdd1 tests/test_zdd1.cpp)
target_link_libraries(test_zdd1 morris_zdd)
# test_board dumps the compiled board specs to config/*.json, so it must run from the
# source root where that directory lives.
add_test(NAME board COMMAND test_board WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
add_executable(test_counts tests/test_counts.cpp)
target_link_libraries(test_counts morris_zdd pthread)
add_test(NAME zdd1 COMMAND test_zdd1)
add_executable(cli_verify src/cli_verify.cpp)
target_link_libraries(cli_verify morris_zdd pthread)
add_library(morris_solve src/solve.cpp)
target_link_libraries(morris_solve morris_zdd)
add_executable(test_3mm tests/test_3mm.cpp)
target_link_libraries(test_3mm morris_solve pthread)
add_test(NAME counts COMMAND test_counts)
add_executable(morris src/cli_main.cpp)
target_link_libraries(morris morris_solve pthread)
add_test(NAME mm3 COMMAND test_3mm)
